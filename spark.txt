cat > ~/spark_wc.sh <<'SCRIPT'
main() {
set -u
say(){ printf '\n\033[1;36m==> %s\033[0m\n' "$*"; }
ok(){ printf '\033[1;32m[OK]\033[0m %s\n' "$*"; }
die(){ printf '\n\033[1;31m[FAILED]\033[0m %s\n' "$*" >&2; exit 1; }
valid_root(){ [ -x "$1/bin/spark-shell" ] && [ -x "$1/bin/spark-submit" ] && ls "$1"/jars/spark-core_*.jar >/dev/null 2>&1; }
find_spark(){
local c r
while IFS= read -r c; do
case "$c" in *site-packages*|*dist-packages*) continue;; esac
r=$(dirname "$(dirname "$(readlink -f "$c")")")
if valid_root "$r"; then echo "$r"; return 0; fi
done < <(find / \( -path /proc -o -path /sys -o -path /dev -o -path /run -o -path /snap -o -path /var/lib/docker \) -prune -o -name spark-shell \( -type f -o -type l \) -path '*/bin/spark-shell' -print 2>/dev/null)
return 1
}
say "Locating Spark"
SH=""
if [ -n "${SPARK_HOME:-}" ] && valid_root "$SPARK_HOME"; then SH="$SPARK_HOME"; else SH=$(find_spark); fi
[ -n "$SH" ] || die "No valid Spark installation found. Run COMMAND 1 first."
SH=$(readlink -f "$SH")
VER=$(ls "$SH"/jars/spark-core_*.jar | head -1 | sed -E 's/.*spark-core_[0-9.]+-([0-9][0-9.]*)\.jar/\1/')
case "$VER" in 4.*) JL="17 21";; *) JL="17 11 1\.8 21";; esac
java_ok(){ [ -x "$1/bin/java" ] && "$1/bin/java" -version 2>&1 | grep -Eq "version \"$2[.\"_]"; }
find_java(){
local p d
for p in $JL; do
for d in "${JAVA_HOME:-}" /usr/lib/jvm/*; do
if [ -n "$d" ] && java_ok "$d" "$p"; then readlink -f "$d"; return 0; fi
done
done
return 1
}
JH=$(find_java)
[ -n "$JH" ] || die "No Java compatible with Spark $VER found. Run COMMAND 1 first."
export JAVA_HOME="$JH" SPARK_HOME="$SH" SPARK_LOCAL_IP=127.0.0.1
export PATH="$SH/bin:$SH/sbin:$JH/bin:$PATH"
ok "Spark $VER at $SH"
ok "Java at $JH"
say "Creating input file"
D="$HOME/spark_lab"; mkdir -p "$D"
printf 'Radhika studies Spark\nSpark runs Scala\nRadhika learns Scala\n' > "$D/input.txt"
cat "$D/input.txt"
cat > "$D/WordCount.scala" <<'EOF'
val lines = sc.textFile("file://" + sc.getConf.get("spark.lab.input"))
val counts = lines.flatMap(_.split("\\s+")).filter(_.nonEmpty).map(w => (w, 1)).reduceByKey(_ + _).sortByKey()
counts.collect().foreach { case (w, n) => println("RES|" + w + "|" + n) }
sys.exit(0)
EOF
say "Scala WordCount program"
cat "$D/WordCount.scala"
say "Running on Spark (RDD WordCount)"
"$SH/bin/spark-shell" --master 'local[*]' --conf spark.ui.enabled=false --conf spark.driver.bindAddress=127.0.0.1 --conf spark.driver.host=127.0.0.1 --conf spark.lab.input="$D/input.txt" -i "$D/WordCount.scala" </dev/null >"$D/run.log" 2>&1
GOT=$(grep -E '^RES\|[^|]+\|[0-9]+$' "$D/run.log" | LC_ALL=C sort)
if [ -z "$GOT" ]; then tail -n 40 "$D/run.log"; die "Spark job produced no output (log above, full log: $D/run.log)."; fi
say "FINAL OUTPUT (word : count)"
printf '%s\n' "$GOT" | awk -F'|' '{printf "%-10s : %s\n",$2,$3}'
EXP=$(printf 'RES|Radhika|2\nRES|Scala|2\nRES|Spark|2\nRES|learns|1\nRES|runs|1\nRES|studies|1\n' | LC_ALL=C sort)
[ "$GOT" = "$EXP" ] || die "Output does not match the expected counts (full log: $D/run.log)."
printf '\n\033[1;32mSUCCESS: Spark Scala WordCount completed and output matches the expected counts.\033[0m\n'
}
main </dev/null
SCRIPT
bash ~/spark_wc.sh
